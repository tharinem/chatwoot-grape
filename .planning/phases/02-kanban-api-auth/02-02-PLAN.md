import { z } from 'zod';

// Response schema for a single stage
export const stageSchema = z.object({
  id: z.string(),
  accountId: z.number(),
  name: z.string(),
  position: z.number(),
  color: z.string().nullable(),
  createdAt: z.string().datetime(),
  updatedAt: z.string().datetime(),
});

// POST /api/v1/stages body
export const createStageSchema = z.object({
  name: z.string().min(1).max(100),
  color: z.string().regex(/^#[0-9A-Fa-f]{6}$/).optional(),
});

// PATCH /api/v1/stages/:id body
export const updateStageSchema = z.object({
  name: z.string().min(1).max(100).optional(),
  color: z.string().regex(/^#[0-9A-Fa-f]{6}$/).nullable().optional(),
});

// PATCH /api/v1/stages/reorder body
export const reorderStagesSchema = z.object({
  stages: z.array(z.object({
    id: z.string(),
    position: z.number().int().positive(),
  })).min(1),
});

// Param schema for :id
export const stageParamsSchema = z.object({
  id: z.string(),
});

export type CreateStageInput = z.infer<typeof createStageSchema>;
export type UpdateStageInput = z.infer<typeof updateStageSchema>;
export type ReorderStagesInput = z.infer<typeof reorderStagesSchema>;

1. **Create `src/routes/v1/stages.ts`** — export an async function `stageRoutes(fastify: FastifyInstance)`:

   All routes use `onRequest: [fastify.authenticate]`. Write routes (POST, PATCH, DELETE, reorder) additionally check `request.user.role === 'administrator'` and return 403 if not. Use `problemResponse` for errors.

   CRITICAL per TENANT-01/TENANT-02: EVERY Prisma query MUST include `where: { accountId }` from `request.user.account_id`. Never query without it.

   **GET /stages** (all roles):
   ```typescript
   const { account_id } = request.user as JwtPayload;
   const stages = await prisma.stage.findMany({
     where: { accountId: account_id },
     orderBy: { position: 'asc' },
     include: { _count: { select: { cards: { where: { deletedAt: null } } } } },
   });
   return stages;
   ```

   **POST /stages** (admin only):
   - Body validated by `createStageSchema`
   - Determine next position: `const maxPos = await prisma.stage.aggregate({ where: { accountId: account_id }, _max: { position: true } }); const nextPos = (maxPos._max.position ?? 0) + 1;`
   - Create: `prisma.stage.create({ data: { accountId: account_id, name: body.name, position: nextPos, color: body.color ?? null } })`
   - Return 201 with created stage

   **PATCH /stages/reorder** (admin only) — MUST be registered BEFORE `/stages/:id` to avoid route conflict:
   - Body validated by `reorderStagesSchema`
   - Use `prisma.$transaction()` with an array of `prisma.stage.update()` calls, each with `where: { id: s.id, accountId: account_id }` and `data: { position: s.position }` (per research Pattern: Stage Reorder Endpoint)
   - The `accountId` in the WHERE clause prevents cross-tenant manipulation (TENANT-02)
   - Return updated stages list

   **PATCH /stages/:id** (admin only):
   - Params validated by `stageParamsSchema`, body by `updateStageSchema`
   - `prisma.stage.update({ where: { id: params.id, accountId: account_id }, data: { ...body } })`
   - If Prisma throws RecordNotFound, return 404
   - Return updated stage

   **DELETE /stages/:id** (admin only):
   - Check if stage has any non-deleted cards: `prisma.card.count({ where: { stageId: params.id, accountId: account_id, deletedAt: null } })`
   - If cards exist, return 409 Conflict with detail "Stage has active cards. Move or delete them first."
   - Otherwise: `prisma.stage.delete({ where: { id: params.id, accountId: account_id } })`
   - Return 204 No Content

   Each route definition should include `schema: { body/params/response }` with the Zod schemas for Swagger auto-generation (per API-04).

2. **Update `src/app.ts`:** Import and register `stageRoutes` under the `/api/v1` prefix alongside `authRoutes`.
